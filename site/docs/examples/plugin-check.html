Local executable adapter fixture

From the repository root:

go build -o /tmp/agenthealth-probe ./examples/plugin-check
python3 - <<'PY'
import hashlib, json
from pathlib import Path
binary = Path('/tmp/agenthealth-probe').resolve()
manifest = {'protocol': 'v1', 'command': str(binary),
            'sha256': hashlib.sha256(binary.read_bytes()).hexdigest(),
            'metadata': {'name': 'protocol-fixture', 'version': '1.0.0',
                         'compatibility_version': 'v1', 'target_types': ['custom'],
                         'dimensions': ['configuration', 'reachability'], 'active_checks': []}}
Path('/tmp/agenthealth-adapter.json').write_text(json.dumps(manifest))
PY
agenthealth check examples/plugin-check/agenthealth.yaml --adapter /tmp/agenthealth-adapter.json --format json

The fixture returns HEALTHY for configuration and UNKNOWN for reachability: it performs no network operation and makes no remote health claim. Expected health exit code is 5. Adapt the output path for your OS; Windows uses an .exe suffix.

The direct/composite names identify agent roles. Dependencies record independently observed health, not verified communication. Add an explicit peer/path target to measure that path. Keep built-in gateway/router, A2A and backend checks separate; see the extension guide. Do not forward arbitrary input, log credentials or return response bodies. Real probes must implement the safety and response-marker rules in the contract.

Configuration files

agenthealth.yaml

version: v1
targets:
  - name: direct-agent-extension-fixture
    type: custom
    endpoint: https://direct-agent.example.invalid/health
    checks: [configuration, reachability]
    dependencies:
      - name: composite-agent-extension-fixture
        type: custom
        endpoint: https://composite-agent.example.invalid/health
        checks: [configuration, reachability]

main.go

// This executable demonstrates the v1 wire protocol without making network calls.
// It is a protocol fixture, not evidence of a real remote service's health.
package main

import (
	"encoding/json"
	"github.com/TheAgentHealth/agenthealth/adapters/process"
	"github.com/TheAgentHealth/agenthealth/core"
	"os"
)

func main() {
	var request process.WireRequest
	decoder := json.NewDecoder(os.Stdin)
	decoder.DisallowUnknownFields()
	if decoder.Decode(&request) != nil || request.Protocol != "v1" {
		os.Exit(1)
	}
	status := core.Healthy
	if request.Dimension != "configuration" {
		status = core.Unknown
	}
	if json.NewEncoder(os.Stdout).Encode(process.Event{Kind: "result", Status: status}) != nil {
		os.Exit(1)
	}
}
